图1 校园委托系统 E-R 图（Crow's Foot / IE 记法）

矩形 = 实体，属性列在框内；PK 主键 / UK 唯一；连线两端鸦爪符号表示基数。
基数记号：|| 一（强制）　|o 零或一　}| 一或多　}o 零或多

erDiagram
    USER     ||--o| WALLET      : "拥有"
    USER     ||--o{ TASK        : "发布"
    CATEGORY ||--o{ TASK        : "所属分类"
    TASK     ||--|| ORDERS      : "成交"
    TASK     ||--o{ APPLICATION : "收到申请"
    ORDERS   ||--o| REVIEW      : "产生评价"
    ORDERS   ||--o{ MESSAGE     : "订单沟通"
    WALLET   ||--o{ TRANSACTION : "产生流水"

    USER {
        bigint   user_id       PK "用户ID"
        varchar  username      UK "登录名"
        char     password_hash    "密码哈希"
        varchar  student_no    UK "学号"
        varchar  phone         UK "手机号"
        varchar  real_name        "真实姓名"
        tinyint  verify_status    "认证状态"
        int      credit_score     "信用分"
        tinyint  status           "账号状态"
        datetime created_at       "注册时间"
    }
    CATEGORY {
        bigint  category_id PK "分类ID"
        varchar name        UK "分类名称"
        varchar description    "分类说明"
        int     sort_no        "排序号"
        tinyint enabled        "是否启用"
    }
    TASK {
        bigint   task_id      PK "委托ID"
        bigint   publisher_id FK "发布者ID"
        bigint   category_id  FK "分类ID"
        varchar  title           "委托标题"
        text     description     "详细描述"
        decimal  reward          "悬赏金额"
        datetime deadline        "期望完成时间"
        varchar  place           "地点"
        tinyint  urgency         "紧急程度"
        tinyint  status          "委托状态"
        int      view_count      "浏览量"
        datetime created_at      "发布时间"
    }
    APPLICATION {
        bigint   application_id PK "申请ID"
        bigint   task_id        FK "委托ID"
        bigint   applicant_id   FK "申请人ID"
        varchar  message           "申请留言"
        tinyint  status            "申请状态"
        datetime applied_at        "申请时间"
        datetime handled_at        "处理时间"
    }
    ORDERS {
        bigint   order_id     PK "订单ID"
        varchar  order_no     UK "订单编号"
        bigint   task_id      FK "委托ID(唯一)"
        bigint   publisher_id FK "委托方ID"
        bigint   taker_id     FK "接单方ID"
        decimal  amount          "成交金额"
        tinyint  status          "订单状态"
        datetime accepted_at     "接单时间"
        datetime finished_at     "完成时间"
    }
    REVIEW {
        bigint   review_id   PK "评价ID"
        bigint   order_id    FK "订单ID(唯一)"
        bigint   reviewer_id FK "评价人ID"
        bigint   reviewee_id FK "被评价人ID"
        tinyint  score          "评分1~5"
        varchar  content        "评价内容"
        datetime created_at     "评价时间"
    }
    WALLET {
        bigint   wallet_id  PK "钱包ID"
        bigint   user_id    FK "用户ID(唯一)"
        decimal  balance       "可用余额"
        decimal  frozen         "冻结金额"
        datetime updated_at     "更新时间"
    }
    TRANSACTION {
        bigint   txn_id        PK "流水ID"
        bigint   wallet_id     FK "钱包ID"
        bigint   order_id      FK "订单ID"
        tinyint  type             "流水类型"
        decimal  amount           "变动金额"
        decimal  balance_after    "变动后余额"
        datetime created_at       "发生时间"
    }
    MESSAGE {
        bigint   message_id  PK "消息ID"
        bigint   order_id    FK "订单ID"
        bigint   sender_id   FK "发送者ID"
        bigint   receiver_id FK "接收者ID"
        varchar  content        "消息内容"
        tinyint  is_read        "是否已读"
        datetime sent_at        "发送时间"
    }

怎么改：用记事本或 VS Code 打开本文件，找到对应标签页里的 erDiagram 或 flowchart LR 段落直接编辑，保存后刷新浏览器即可。
怎么导出 PNG / SVG：把对应源码整段复制到 mermaid.live，右上角 Actions → PNG / SVG。
渲染不出来？先确认已联网（Mermaid 库从 CDN 加载）。若断网，请改用 生成ER图.bat 出 PNG，或 11_ER图源码.md 里的 Graphviz DOT 方案。
